<script>
    // 步驟 1: FHIR 轉譯與黑名單去識別化
    function processFHIR() {
        const fhirData = {
            resourceType: "Bundle",
            id: "bundle-emergency-" + Math.floor(Math.random() * 1000),
            entry: [
                {
                    resource: {
                        resourceType: "Patient",
                        id: "UUID-8a9f-3b7c",
                        name: "[MASKED_BY_BLOCKLIST]",
                        identifier: "[MASKED_BY_BLOCKLIST]"
                    }
                },
                {
                    resource: {
                        resourceType: "Condition",
                        code: document.getElementById('p-diag').value
                    }
                },
                {
                    resource: {
                        resourceType: "Observation",
                        value: document.getElementById('p-obs').value
                    }
                }
            ]
        };
        document.getElementById('fhir-json').innerText = JSON.stringify(fhirData, null, 2);
        document.getElementById('btn-upload').style.display = 'inline-flex';
    }

    // 切換畫面步驟
    function goToStage(stageNum) {
        document.querySelectorAll('.stage-card').forEach(el => el.classList.remove('show'));
        document.querySelectorAll('.step-item').forEach(el => el.classList.remove('active'));

        document.getElementById(`stage-${stageNum}`).classList.add('show');
        document.getElementById(`step-tab-${stageNum}`).classList.add('active');

        for (let i = 1; i < stageNum; i++) {
            document.getElementById(`step-tab-${i}`).classList.add('completed');
        }

        if (stageNum === 2) {
            appendLog("[TRANSMIT] FHIR Bundle uploaded from Hospital Gateway (192.168.1.104).");
            appendLog("[AUDIT] Record #88214 written to immutable Audit Log.");
            appendLog("[TRIGGER] Critical ICD-10 'S72.0' identified. WebSocket ready.");
        }
    }

    function appendLog(text) {
        const box = document.getElementById('audit-log');
        const time = new Date().toLocaleTimeString();
        box.innerHTML += `\n[${time}] ${text}`;
        box.scrollTop = box.scrollHeight;
    }

    // 步驟 2 ➔ 步驟 3: 觸發推播
    function triggerPushNotification() {
        const alertBox = document.getElementById('alert-box');
        alertBox.style.display = 'block';
        setTimeout(() => { alertBox.style.display = 'none'; }, 3500);
        goToStage(3);
    }

    // 步驟 3: 金鑰還原身分
    function decryptIdentity() {
        const key = document.getElementById('auth-key').value;
        const card = document.getElementById('patient-card');
        if (key === "1234") {
            card.innerHTML = `
                <div style="color: #4ade80; font-weight: bold; margin-bottom: 8px;">✔ 憑證解密成功！身分還原對照完成</div>
                <p><b>姓名：</b> 張阿嬤 (80 歲)</p>
                <p><b>身分證字號：</b> A287654321</p>
                <p><b>聯絡電話：</b> 0912-345-678</p>
                <p><b>轄區狀態：</b> 萬華區個案 (負責個管員: 王專員)</p>
            `;
            document.getElementById('btn-to-ai').style.display = 'inline-flex';
        } else {
            alert('金鑰錯誤！(請輸入測試金鑰: 1234)');
        }
    }

    // 步驟 4: AI 打字機效果生成計畫
    function generateAIPlan() {
        const planText = `【AI CDSS 出院準備與連續照顧建議書】\n
▶ 評估等級：高風險長輩 (股骨頸骨折 / 高血壓)\n
▶ 建議銜接資源：
  1. 復健服務：出院 48 小時內啟動社區居家物理治療 (PT) 評估。
  2. 環境評估：申請住家防跌無障礙扶手與防滑設施。
  3. 居服派案：術後第一個月安排居家服務員協助日常起居。\n
▶ 黃金 24 小時目標：已將推薦表單同步匯入個管員派案庫。`;

        const box = document.getElementById('ai-output');
        box.innerText = "";
        let i = 0;
        const timer = setInterval(() => {
            box.innerText += planText.charAt(i);
            i++;
            if (i >= planText.length) {
                clearInterval(timer);
                document.getElementById('btn-complete').style.display = 'inline-flex';
            }
        }, 20);
    }
